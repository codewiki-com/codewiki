<Sheet title="Install and authenticate">
  <Row code="npm install -g @openai/codex@0.153.0">install the exact CLI release used by this sheet</Row>
  <Row code="codex --version">print the installed CLI version</Row>
  <Row code="codex login">authenticate through the default browser flow</Row>
  <Row code="codex login --device-auth">authenticate with a device code instead of opening a browser</Row>
  <Row code="printenv OPENAI_API_KEY | codex login --with-api-key">read an API key from standard input</Row>
  <Row code="codex login status">show the active authentication mode and return nonzero when logged out</Row>
  <Row code="codex logout">remove stored Codex credentials</Row>
</Sheet>

<Sheet title="Start an interactive task">
  <Row code="codex">open the terminal UI in the current directory</Row>
  <Row code="codex 'Explain the failing test'">open the terminal UI with an initial task</Row>
  <Row code="codex -C repo 'Fix the failing test'">set the working root before the task begins</Row>
  <Row code="codex -i screenshot.png 'Fix this UI bug'">attach an image to the initial task</Row>
  <Row code="codex --search 'Verify the current dependency API'">enable live web search for this run</Row>
  <Row code="codex --no-alt-screen">keep the terminal UI in normal scrollback</Row>
</Sheet>

<Sheet title="Bound permissions">
  <Row code="codex -s read-only">allow model-generated commands to read but not write</Row>
  <Row code="codex -s workspace-write -a on-request">allow workspace writes and let Codex request approval</Row>
  <Row code="codex -s workspace-write -a never">allow workspace writes without interactive approval prompts</Row>
  <Row code="codex --add-dir ../shared">grant another directory write access for this run</Row>
  <Row code="codex --approve-for-me">route approval requests through automatic review in workspace-write mode</Row>
  <Row code="codex --dangerously-bypass-approvals-and-sandbox">remove both safeguards only inside an externally hardened sandbox</Row>
</Sheet>

<Sheet title="Control the terminal UI">
  <Row code="/init">generate an AGENTS.md scaffold for the current project</Row>
  <Row code="/status">show the active session configuration and usage</Row>
  <Row code="/model">choose the model and reasoning effort</Row>
  <Row code="/permissions">change the approval and sandbox preset during the session</Row>
  <Row code="/mention src/lib/api.ts">add a specific workspace file to the prompt</Row>
  <Row code="/compact">replace earlier turns with a concise context summary</Row>
  <Row code="/exit">close the interactive session</Row>
</Sheet>

<Sheet title="Run non-interactively">
  <Row code="codex exec 'Run tests and fix failures'">run one task without opening the terminal UI</Row>
  <Row code="printf '%s\n' 'Explain this diff' | codex exec -">read the task from standard input</Row>
  <Row code="codex exec --ephemeral 'Explain this repository'">run without persisting session files</Row>
  <Row code="codex exec --json 'Run the targeted tests'">stream machine-readable events as JSONL</Row>
  <Row code="codex exec -o final.txt 'Summarize the change'">write the final agent message to a file</Row>
  <Row code="codex exec --output-schema schema.json 'Report findings'">constrain the final response with a JSON Schema</Row>
</Sheet>

<Sheet title="Continue or branch a session">
  <Row code="codex resume">choose a saved interactive session from the current directory</Row>
  <Row code="codex resume --last">continue the most recent saved interactive session</Row>
  <Row code="codex resume SESSION_ID">continue one saved session by identifier or name</Row>
  <Row code="codex resume --all">show sessions outside the current-directory filter</Row>
  <Row code="codex fork SESSION_ID">start a new chat from an earlier session</Row>
  <Row code="codex fork --last">fork the most recent saved interactive session</Row>
  <Row code="codex exec resume --last 'Run the remaining checks'">continue the latest non-interactive session with a new task</Row>
</Sheet>

<Sheet title="Review changes">
  <Row code="/diff">show staged, unstaged, and untracked workspace changes</Row>
  <Row code="/review">review the working tree inside the current session</Row>
  <Row code="codex review --uncommitted">review staged, unstaged, and untracked changes non-interactively</Row>
  <Row code="codex review --base main">review the current branch against main</Row>
  <Row code="codex review --commit COMMIT">review the changes introduced by one commit</Row>
  <Row code="printf '%s\n' 'Focus on correctness and missing tests' | codex review -">read custom review instructions from standard input</Row>
</Sheet>

<Sheet title="Override configuration">
  <Row code="codex -m MODEL">override the configured model for this run</Row>
  <Row code="codex -p PROFILE">layer a named profile over the base configuration</Row>
  <Row code="codex -c key=value">override one configuration value for this invocation</Row>
  <Row code="codex --strict-config">fail when config.toml contains an unknown field</Row>
  <Row code="codex features list">show known feature flags, stages, and effective states</Row>
  <Row code="codex features enable FEATURE">persistently enable one feature in config.toml</Row>
  <Row code="codex features disable FEATURE">persistently disable one feature in config.toml</Row>
</Sheet>

<Sheet title="Manage MCP servers">
  <Row code="codex mcp list">list configured Model Context Protocol servers</Row>
  <Row code="codex mcp list --json">list configured servers as JSON</Row>
  <Row code="codex mcp get NAME --json">show one server configuration as JSON</Row>
  <Row code="codex mcp add NAME -- COMMAND ARG">register a server launched over standard input and output</Row>
  <Row code="codex mcp add NAME --url URL">register a server reached through streamable HTTP</Row>
  <Row code="codex mcp login NAME">authenticate one HTTP server with OAuth</Row>
  <Row code="codex mcp remove NAME">remove one server configuration</Row>
</Sheet>

<Sheet title="Diagnose and integrate">
  <Row code="codex doctor">inspect installation, configuration, authentication, runtime, Git, and terminal health</Row>
  <Row code="codex doctor --summary">print only grouped diagnostic checks and totals</Row>
  <Row code="codex doctor --json">emit a redacted machine-readable diagnostic report</Row>
  <Row code="codex completion bash">generate a Bash completion script</Row>
  <Row code="codex sandbox -- COMMAND ARG">run an arbitrary command inside the Codex-provided sandbox</Row>
  <Row code="codex update">check for and apply a supported CLI self-update</Row>
</Sheet>
