<Sheet title="值">
  <Row code="const value = source">绑定不可重新赋值的块级名称</Row>
  <Row code="let value = source">绑定可以重新赋值的块级名称</Row>
  <Row code="value ?? fallback">仅当值为 `null` 或 `undefined` 时使用后备值</Row>
  <Row code="items.at(-1)">读取最后一项且不修改数组</Row>
  <Row code="Object.hasOwn(data, key)">明确检查对象自身是否有该属性</Row>
</Sheet>

<Sheet title="函数">
  <Row code="const f = (x) => x + 1">从箭头函数返回一个表达式</Row>
  <Row code="function f(x) { return x + 1; }">声明具名函数</Row>
  <Row code="f(...values)">把可迭代对象展开为参数</Row>
  <Row code="f.call(owner, value)">用指定的接收者调用函数</Row>
  <Row code="f.bind(owner)">创建绑定了接收者的函数</Row>
</Sheet>

<Sheet title="集合">
  <Row code="items.map(project)">逐项转换并生成新数组</Row>
  <Row code="items.filter(keep)">用匹配项生成新数组</Row>
  <Row code="items.find(match)">返回第一个匹配项，未找到则返回 `undefined`</Row>
  <Row code="items.some(match)">检查是否有任意一项匹配</Row>
  <Row code="items.every(match)">检查是否所有项都匹配</Row>
</Sheet>

<Sheet title="对象">
  <Row code="{ ...base, key: value }">浅拷贝对象并覆盖一个属性</Row>
  <Row code="Object.entries(data)">读取对象自身可枚举的字符串键条目</Row>
  <Row code="Object.fromEntries(rows)">从键值对构建对象</Row>
  <Row code="delete data.key">删除对象自身可配置的属性</Row>
  <Row code="structuredClone(data)">深拷贝受支持的结构化数据</Row>
</Sheet>

<Sheet title="错误">
  <Row code="throw new Error(message)">显式抛出错误</Row>
  <Row code="try { work(); } catch (error) { handle(error); }">处理同步异常</Row>
  <Row code="promise.catch(handle)">处理被拒绝的 Promise</Row>
  <Row code="finally { release(); }">无论结果如何都释放资源</Row>
  <Row code="error instanceof TypeError">检查错误是否属于已知类型</Row>
</Sheet>

<Sheet title="异步">
  <Row code="await task">等待一个 Promise 敲定</Row>
  <Row code="await Promise.all(tasks)">收集全部结果，或在首次失败时拒绝</Row>
  <Row code="await Promise.allSettled(tasks)">收集每个兑现与拒绝结果</Row>
  <Row code="AbortSignal.timeout(ms)">创建超时后自动中止的信号</Row>
  <Row code="for await (const item of source)">依次消费异步可迭代对象</Row>
</Sheet>

<Sheet title="模块">
  <Row code="export { value }">导出具名绑定</Row>
  <Row code="import { value } from './mod.js'">导入具名绑定</Row>
  <Row code="export default value">导出一个默认值</Row>
  <Row code="import('./mod.js')">动态加载模块并返回 Promise</Row>
  <Row code="import.meta.url">读取当前模块的 URL</Row>
</Sheet>

<Sheet title="字符串">
  <Row code="`id: ${id}`">把值插入模板字面量</Row>
  <Row code="text.includes(part)">检查是否包含子字符串</Row>
  <Row code="text.replaceAll(a, b)">替换所有完全匹配的字符串</Row>
  <Row code="text.split(separator)">把文本拆分为数组</Row>
  <Row code="parts.join(separator)">把多个值连接为文本</Row>
</Sheet>

<Sheet title="输入边界">
  <Row code="Number.parseInt(text, 10)">解析开头的十进制整数</Row>
  <Row code="Number.isFinite(value)">不进行类型强制转换，检查是否为有限数</Row>
  <Row code="encodeURIComponent(value)">编码一个 URL 组件</Row>
  <Row code="JSON.parse(text)">解析 JSON 文本</Row>
  <Row code="JSON.stringify(value)">序列化与 JSON 兼容的值</Row>
</Sheet>
