<Sheet title="Bindings and control flow">
  <Row code="let value = source;">bind an immutable local; shadow it with another `let` when the type must change</Row>
  <Row code="let mut count = 0;">permit reassignment and in-place mutation</Row>
  <Row code="const LIMIT: usize = 100;">declare a typed compile-time constant</Row>
  <Row code="let Some(id) = maybe_id else { return; };">extract a matching value or leave the current control flow</Row>
  <Row code="match result { Ok(value) =&gt; value, Err(error) =&gt; return Err(error) }">handle every variant and return the original error</Row>
  <Row code="while let Some(item) = stack.pop() { consume(item); }">keep consuming values until the pattern stops matching</Row>
</Sheet>

<Sheet title="Ownership and borrowing">
  <Row code="inspect(&amp;value);">lend shared access without transferring ownership</Row>
  <Row code="update(&amp;mut value);">lend exclusive mutable access until the borrow's last use</Row>
  <Row code="let moved = value;">transfer ownership and invalidate the old binding unless the type is `Copy`</Row>
  <Row code="let copied = number;">copy a `Copy` value so both bindings remain usable</Row>
  <Row code="let cloned = value.clone();">request an explicit duplicate through `Clone`</Row>
  <Row code="let old = std::mem::take(&amp;mut value);">replace with `Default::default()` and take the old value</Row>
  <Row code="drop(guard);">release an owned resource before the end of its scope</Row>
</Sheet>

<Sheet title="Strings and collections">
  <Row code="fn normalize(text: &amp;str) -&gt; String { text.trim().to_owned() }">borrow UTF-8 input and return owned text</Row>
  <Row code="text.chars()">iterate Unicode scalar values, not grapheme clusters</Row>
  <Row code="text.as_bytes()">borrow the underlying UTF-8 bytes without allocating</Row>
  <Row code="String::from_utf8(bytes)?">validate UTF-8 and reuse the owned byte buffer</Row>
  <Row code="Vec::with_capacity(count)">reserve space when the expected item count is known</Row>
  <Row code="counts.entry(key).or_insert(0)">insert a missing map value and borrow it mutably</Row>
  <Row code="queue.pop_front()">remove the oldest `VecDeque` item as an `Option`</Row>
</Sheet>

<Sheet title="Iterators and closures">
  <Row code="items.iter()">yield shared references and keep the collection</Row>
  <Row code="items.iter_mut()">yield exclusive references for in-place updates</Row>
  <Row code="items.into_iter()">consume the collection and yield owned items</Row>
  <Row code="iter.filter_map(parse).collect::&lt;Vec&lt;_&gt;&gt;()">discard `None` values and collect successful projections</Row>
  <Row code="iter.collect::&lt;Result&lt;Vec&lt;_&gt;, _&gt;&gt;()?">stop at the first error while collecting successful items</Row>
  <Row code="move |item| process(item, &amp;config)">capture `config` by value for a closure that may outlive this scope</Row>
  <Row code="F: FnMut(&amp;Item) -&gt; bool">accept a repeatable predicate that may mutate captured state</Row>
</Sheet>

  <Row code="Box&lt;dyn Draw + Send&gt;">own a type-erased implementor that can cross thread boundaries</Row>
  <Row code="&lt;User as Display&gt;::fmt(&amp;user, formatter)">select a trait implementation explicitly with fully qualified syntax</Row>

<Sheet title="Option and Result">
  <Row code="type LoadResult&lt;T&gt; = Result&lt;T, LoadError&gt;;">name a reusable result type without hiding its error</Row>
  <Row code="let value = operation()?;">unwrap success or return a converted residual immediately</Row>
  <Row code="option.ok_or_else(|| Error::Missing(key))?">turn absence into a lazily constructed error</Row>
  <Row code="option.and_then(parse)">chain a function that returns another `Option`</Row>
  <Row code="result.map(transform)">transform only the success value</Row>
  <Row code="result.map_err(Error::from)">convert only the error value</Row>
  <Row code="matches!(status, Status::Ready(_))">test a pattern and return a boolean</Row>
</Sheet>

<Sheet title="anyhow and thiserror">
  <Row code="type AppResult&lt;T&gt; = anyhow::Result&lt;T&gt;;">erase concrete error types at an application boundary</Row>
  <Row code="read(path).with_context(|| format!(&quot;reading {path}&quot;))?">after importing `anyhow::Context`, add context only on failure</Row>
  <Row code="anyhow::bail!(&quot;invalid port: {port}&quot;);">return an ad hoc application error immediately</Row>
  <Row code="anyhow::ensure!(port &gt; 0, &quot;port must be positive&quot;);">return an application error when an invariant is false</Row>
  <Row code="#[derive(Debug, thiserror::Error)] enum LoadError { #[error(&quot;missing config&quot;)] Missing }">define a concrete error that callers can match</Row>
  <Row code="#[error(&quot;cannot read {path}: {source}&quot;)] Read { path: PathBuf, #[source] source: std::io::Error }">format a variant while preserving its lower-level source</Row>
  <Row code="Io(#[from] std::io::Error)">generate `From` conversion and error-source reporting for a variant</Row>
</Sheet>

<Sheet title="Threads and shared state">
  <Row code="let handle = std::thread::spawn(move || work(data));">move owned task data into a new operating-system thread</Row>
  <Row code="let result = handle.join().expect(&quot;worker panicked&quot;);">wait for a thread and surface a panic explicitly</Row>
  <Row code="let shared = Arc::new(Mutex::new(value));">combine shared ownership with synchronized mutation</Row>
  <Row code="let mut guard = shared.lock().unwrap();">acquire mutable access; handle poisoning when recovery matters</Row>
  <Row code="let read_guard = lock.read().unwrap();">allow concurrent readers through an `RwLock` guard</Row>
  <Row code="let (sender, receiver) = std::sync::mpsc::sync_channel(capacity);">create a bounded channel whose full buffer applies backpressure</Row>
  <Row code="sender.send(value)?;">transfer a value to the receiver or report disconnection</Row>
</Sheet>

<Sheet title="Async and Tokio">
  <Row code="async fn fetch() -&gt; Result&lt;Data, Error&gt; { request().await }">return a future whose body runs when polled</Row>
  <Row code="let value = future.await?;">suspend until completion and propagate failure</Row>
  <Row code="let handle = tokio::spawn(async move { work(data).await });">spawn an owned `Send + 'static` task on the runtime</Row>
  <Row code="let (left, right) = tokio::try_join!(load_a(), load_b())?;">drive fallible futures concurrently and stop on an error</Row>
  <Row code="tokio::select! { value = task =&gt; value, }">race enabled branches; losing futures must be cancellation-safe</Row>
  <Row code="let value = tokio::time::timeout(duration, operation()).await??;">bound the wait and propagate both timeout and operation errors</Row>
  <Row code="tokio::task::spawn_blocking(|| blocking_work()).await?;">move blocking work off asynchronous worker threads</Row>
</Sheet>
