  <Row code="SELECT c.name, o.order_id FROM customers AS c LEFT JOIN orders AS o ON o.customer_id = c.customer_id;">keep every customer even without an order</Row>
  <Row code="SELECT * FROM colors CROSS JOIN sizes;">produce every color-size combination</Row>
  <Row code="SELECT * FROM orders JOIN customers USING (customer_id);">join on an equally named column and project it once</Row>
  <Row code="SELECT * FROM customers AS c WHERE EXISTS (SELECT 1 FROM orders AS o WHERE o.customer_id = c.customer_id);">keep customers that have at least one order</Row>
  <Row code="SELECT * FROM customers AS c WHERE NOT EXISTS (SELECT 1 FROM orders AS o WHERE o.customer_id = c.customer_id);">keep customers with no matching order, even when the inner key is nullable</Row>

<Sheet title="Stage and recurse">

  <Row code="WITH paid AS (SELECT * FROM orders WHERE status = 'paid'), totals AS (SELECT customer_id, SUM(total) AS spent FROM paid GROUP BY customer_id) SELECT * FROM totals;">feed one CTE into the next</Row>
  <Row code="WITH order_totals AS (SELECT customer_id, SUM(total) AS spent FROM orders GROUP BY customer_id) SELECT c.name, t.spent FROM order_totals AS t JOIN customers AS c USING (customer_id);">join a summarized CTE to detail data</Row>
  <Row code="WITH RECURSIVE sequence(n) AS (VALUES (1) UNION ALL SELECT n + 1 FROM sequence WHERE n < 5) SELECT n FROM sequence;">generate integers one through five with a stopping condition</Row>
  <Row code="WITH RECURSIVE tree(id, depth) AS (SELECT category_id, 0 FROM categories WHERE parent_id IS NULL UNION ALL SELECT c.category_id, t.depth + 1 FROM categories AS c JOIN tree AS t ON c.parent_id = t.id) SELECT * FROM tree;">walk a category hierarchy from its roots</Row>
  <Row code="VALUES (1, 'draft'), (2, 'paid');">construct a two-row, two-column result directly</Row>

<Sheet title="Calculate over windows">
  <Row code="SELECT order_id, ROW_NUMBER() OVER (PARTITION BY customer_id ORDER BY created_at, order_id) AS position FROM orders;">assign a stable position inside each customer partition</Row>
  <Row code="SELECT order_id, RANK() OVER (ORDER BY total DESC) AS place FROM orders;">give equal totals the same rank and leave gaps</Row>
  <Row code="SELECT order_id, DENSE_RANK() OVER (ORDER BY total DESC) AS tier FROM orders;">give equal totals the same rank without gaps</Row>
  <Row code="SELECT order_id, LAG(total) OVER (PARTITION BY customer_id ORDER BY created_at, order_id) AS previous_total FROM orders;">read the preceding ordered value in each partition</Row>
  <Row code="SELECT order_id, LEAD(total) OVER (PARTITION BY customer_id ORDER BY created_at, order_id) AS next_total FROM orders;">read the following ordered value in each partition</Row>
  <Row code="SELECT order_id, SUM(total) OVER (PARTITION BY customer_id ORDER BY created_at, order_id ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS running_total FROM orders;">calculate a running total with an explicit row frame</Row>
</Sheet>
