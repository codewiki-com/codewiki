<Sheet title="安装与认证">
  <Row code="npm install -g @openai/codex@0.153.0">安装本速查表采用的准确CLI版本</Row>
  <Row code="codex --version">输出已安装的CLI版本</Row>
  <Row code="codex login">通过默认浏览器流程认证</Row>
  <Row code="codex login --device-auth">使用设备代码认证，不打开浏览器</Row>
  <Row code="printenv OPENAI_API_KEY | codex login --with-api-key">从标准输入读取API密钥</Row>
  <Row code="codex login status">显示当前认证方式；未登录时返回非零状态</Row>
  <Row code="codex logout">删除已存储的Codex凭据</Row>
</Sheet>

<Sheet title="启动交互任务">
  <Row code="codex">在当前目录打开终端界面</Row>
  <Row code="codex 'Explain the failing test'">打开终端界面并提供初始任务</Row>
  <Row code="codex -C repo 'Fix the failing test'">在任务开始前设置工作根目录</Row>
  <Row code="codex -i screenshot.png 'Fix this UI bug'">为初始任务附加图片</Row>
  <Row code="codex --search 'Verify the current dependency API'">为本次运行启用实时网页搜索</Row>
  <Row code="codex --no-alt-screen">在普通终端回滚区中显示界面</Row>
</Sheet>

<Sheet title="限定权限">
  <Row code="codex -s read-only">允许模型生成的命令读取，但不允许写入</Row>
  <Row code="codex -s workspace-write -a on-request">允许写入工作区，并让Codex按需请求批准</Row>
  <Row code="codex -s workspace-write -a never">允许写入工作区，不显示交互式批准提示</Row>
  <Row code="codex --add-dir ../shared">在本次运行中授予另一目录写权限</Row>
  <Row code="codex --approve-for-me">在workspace-write模式下通过自动审查处理批准请求</Row>
  <Row code="codex --dangerously-bypass-approvals-and-sandbox">仅在外部已加固的沙箱中移除两层防护</Row>
</Sheet>

<Sheet title="控制终端界面">
  <Row code="/init">为当前项目生成AGENTS.md框架</Row>
  <Row code="/status">显示当前会话配置与用量</Row>
  <Row code="/model">选择模型与推理强度</Row>
  <Row code="/permissions">在会话中更改批准与沙箱预设</Row>
  <Row code="/mention src/lib/api.ts">把指定工作区文件加入提示</Row>
  <Row code="/compact">用精简的上下文摘要替换早先轮次</Row>
  <Row code="/exit">关闭交互会话</Row>
</Sheet>

<Sheet title="非交互运行">
  <Row code="codex exec 'Run tests and fix failures'">不打开终端界面，直接运行一个任务</Row>
  <Row code="printf '%s\n' 'Explain this diff' | codex exec -">从标准输入读取任务</Row>
  <Row code="codex exec --ephemeral 'Explain this repository'">运行任务，不在磁盘保存会话文件</Row>
  <Row code="codex exec --json 'Run the targeted tests'">以JSONL流式输出机器可读事件</Row>
  <Row code="codex exec -o final.txt 'Summarize the change'">把智能体最终消息写入文件</Row>
  <Row code="codex exec --output-schema schema.json 'Report findings'">用JSON Schema约束最终响应</Row>
</Sheet>

<Sheet title="继续或分支会话">
  <Row code="codex resume">从当前目录的已保存交互会话中选择</Row>
  <Row code="codex resume --last">继续最近保存的交互会话</Row>
  <Row code="codex resume SESSION_ID">按标识符或名称继续指定会话</Row>
  <Row code="codex resume --all">显示不受当前目录筛选的会话</Row>
  <Row code="codex fork SESSION_ID">从先前会话开始新对话</Row>
  <Row code="codex fork --last">分支最近保存的交互会话</Row>
  <Row code="codex exec resume --last 'Run the remaining checks'">向最近的非交互会话发送新任务并继续运行</Row>
</Sheet>

<Sheet title="审查修改">
  <Row code="/diff">显示已暂存、未暂存和未跟踪的工作区修改</Row>
  <Row code="/review">在当前会话中审查工作树</Row>
  <Row code="codex review --uncommitted">以非交互方式审查已暂存、未暂存和未跟踪的修改</Row>
  <Row code="codex review --base main">审查当前分支相对main的修改</Row>
  <Row code="codex review --commit COMMIT">审查一个提交引入的修改</Row>
  <Row code="printf '%s\n' 'Focus on correctness and missing tests' | codex review -">从标准输入读取自定义审查要求</Row>
</Sheet>

<Sheet title="覆盖配置">
  <Row code="codex -m MODEL">在本次运行中覆盖已配置的模型</Row>
  <Row code="codex -p PROFILE">在基础配置上叠加具名配置档</Row>
  <Row code="codex -c key=value">在本次调用中覆盖一个配置值</Row>
  <Row code="codex --strict-config">config.toml含未知字段时失败</Row>
  <Row code="codex features list">显示已知功能标志、阶段与实际状态</Row>
  <Row code="codex features enable FEATURE">在config.toml中持久启用一项功能</Row>
  <Row code="codex features disable FEATURE">在config.toml中持久停用一项功能</Row>
</Sheet>

<Sheet title="管理MCP服务器">
  <Row code="codex mcp list">列出已配置的Model Context Protocol服务器</Row>
  <Row code="codex mcp list --json">以JSON列出已配置的服务器</Row>
  <Row code="codex mcp get NAME --json">以JSON显示一个服务器配置</Row>
  <Row code="codex mcp add NAME -- COMMAND ARG">注册通过标准输入与输出启动的服务器</Row>
  <Row code="codex mcp add NAME --url URL">注册通过可流式HTTP访问的服务器</Row>
  <Row code="codex mcp login NAME">使用OAuth认证一个HTTP服务器</Row>
  <Row code="codex mcp remove NAME">删除一个服务器配置</Row>
</Sheet>

<Sheet title="诊断与集成">
  <Row code="codex doctor">检查安装、配置、认证、运行时、Git与终端状态</Row>
  <Row code="codex doctor --summary">只输出分组诊断检查与合计</Row>
  <Row code="codex doctor --json">输出已脱敏的机器可读诊断报告</Row>
  <Row code="codex completion bash">生成Bash命令补全脚本</Row>
  <Row code="codex sandbox -- COMMAND ARG">在Codex提供的沙箱中运行任意命令</Row>
  <Row code="codex update">检查并执行当前版本支持的CLI自更新</Row>
</Sheet>
