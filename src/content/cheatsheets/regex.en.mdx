<Sheet title="Literal text">
  <Row code="/cat/">match the literal sequence `cat` anywhere in the input</Row>
  <Row code="/a\.b/">escape the dot so it matches a literal `.`</Row>
  <Row code="/https?:\/\//">match `http://` or `https://`</Row>
  <Row code="/\n/">match one line-feed character</Row>
  <Row code="/\x41/">match `A` through a two-digit hexadecimal escape</Row>
  <Row code="/\u{1F600}/u">match code point `U+1F600` in Unicode mode</Row>
</Sheet>

<Sheet title="Character sets">
  <Row code="/[abc]/">match one listed UTF-16 code unit</Row>
  <Row code="/[^abc]/">match one UTF-16 code unit not in the set</Row>
  <Row code="/[a-z]/">match one lowercase ASCII letter</Row>
  <Row code="/\d/">match one ASCII digit</Row>
  <Row code="/\w/">match one ASCII letter, digit, or underscore</Row>
  <Row code="/\s/">match one ECMAScript whitespace or line-terminator character</Row>
  <Row code="/\p{Letter}/u">match one Unicode code point with the `Letter` property</Row>
</Sheet>

<Sheet title="Quantifiers">
  <Row code="/a?/">match zero or one `a`, preferring one</Row>
  <Row code="/a*/">match zero or more `a` characters greedily</Row>
  <Row code="/a+/">match one or more `a` characters greedily</Row>
  <Row code="/a{3}/">match exactly three `a` characters</Row>
  <Row code="/a{2,}/">match at least two `a` characters</Row>
  <Row code="/a{2,4}/">match two through four `a` characters, preferring four</Row>
  <Row code="/a+?/">match as few `a` characters as the remaining pattern allows</Row>
</Sheet>

<Sheet title="Positions and boundaries">
  <Row code="/^text/">match `text` only at the start of the input</Row>
  <Row code="/text$/">match `text` at the input end or before a final line terminator</Row>
  <Row code="/^text/m">match `text` at the input start or after a line terminator</Row>
  <Row code="/text$/m">match `text` at the input end or before a line terminator</Row>
  <Row code="/\bword\b/">require ASCII word boundaries around `word`</Row>
  <Row code="/\Bword/">require no ASCII word boundary before `word`</Row>
</Sheet>

<Sheet title="Groups and backreferences">
  <Row code="/(cat|dog)/">choose either alternative and capture it as group 1</Row>
  <Row code="/(?:cat|dog)/">group alternatives without creating a capture</Row>
  <Row code="/(?&lt;kind>cat|dog)/">capture the chosen alternative as `groups.kind`</Row>
  <Row code="/((a)b)/">number nested captures by their opening parentheses</Row>
  <Row code="/([a-z]+)-\1/">require the text after `-` to repeat group 1 exactly</Row>
  <Row code="/(?&lt;word>[a-z]+)\s+\k&lt;word>/i">find a repeated ASCII word with a named backreference</Row>
</Sheet>

<Sheet title="Lookaround">
  <Row code="/q(?=u)/">match `q` only when `u` follows, without consuming `u`</Row>
  <Row code="/q(?!u)/">match `q` only when `u` does not follow</Row>
  <Row code="/(?&lt;=\$)\d+/">match digits preceded by `$` without consuming `$`</Row>
  <Row code="/(?&lt;!\$)\b\d+/">match ASCII digits not immediately preceded by `$`</Row>
  <Row code="/^(?!.*forbidden).*$/s">reject any input containing the literal text `forbidden`</Row>
  <Row code="/(?&lt;!\w)cat(?!\w)/">match `cat` outside adjacent ASCII word characters</Row>
</Sheet>

<Sheet title="Flags">
  <Row code="/cat/i">match letters without case sensitivity</Row>
  <Row code="/^cat$/m">make `^` and `$` recognize line boundaries</Row>
  <Row code="/a.b/s">let dot match line terminators</Row>
  <Row code="/cat/g">enable successive matches and stateful `lastIndex`</Row>
  <Row code="/cat/d">include start-end index pairs in match results</Row>
  <Row code="/\p{Script=Greek}+/u">enable Unicode matching and property escapes</Row>
  <Row code="/[\p{Letter}&amp;&amp;\p{Script=Greek}]+/v">use Unicode Sets mode to intersect two properties</Row>
</Sheet>

<Sheet title="Match results">
  <Row code="/cat/.test(text)">return a boolean indicating whether a match exists</Row>
  <Row code="/(?&lt;id>\d+)/.exec(text)">return the first detailed match with a named capture, or `null`</Row>
  <Row code="text.match(/\d+/)">return the first detailed match, or `null`</Row>
  <Row code="text.match(/\d+/g)">return all full matches without capture details, or `null`</Row>
  <Row code="[...text.matchAll(/(?&lt;id>\d+)/g)]">collect detailed results for every match</Row>
  <Row code="match.groups.id">read the value of the named capture `id`</Row>
  <Row code="match.indices[0]">read the full match's `[start, end]` pair when using `d`</Row>
</Sheet>

<Sheet title="Search, split, and replace">
  <Row code="text.search(/error/i)">return the first match index, or `-1`</Row>
  <Row code="text.split(/\s*,\s*/)">split on commas and discard adjacent whitespace</Row>
  <Row code="text.replace(/\s+/g, ' ')">replace every whitespace run with one space</Row>
  <Row code="text.replace(/(\d{4})-(\d{2})-(\d{2})/, '$3/$2/$1')">reorder three numbered captures in the first match</Row>
  <Row code="text.replace(/(?&lt;last>\w+), (?&lt;first>\w+)/, '$&lt;first> $&lt;last>')">reorder named captures in the first match</Row>
  <Row code="text.replace(/\w+/g, word => word.toUpperCase())">compute a replacement for each ASCII word</Row>
</Sheet>

<Sheet title="Construction and state">
  <Row code="RegExp.escape(input)">escape runtime text for literal insertion into a pattern</Row>
  <Row code="new RegExp(RegExp.escape(input), 'u')">build a Unicode-mode pattern that treats `input` literally</Row>
  <Row code="pattern.source">read the pattern text without delimiters or flags</Row>
  <Row code="pattern.flags">read enabled flags in canonical order</Row>
  <Row code="pattern.lastIndex">read where the next global or sticky match starts</Row>
  <Row code="pattern.lastIndex = 0">reset a global or sticky pattern before reuse</Row>
  <Row code="new RegExp(pattern.source, pattern.flags)">clone a pattern with `lastIndex` reset to zero</Row>
</Sheet>

<Sheet title="Text checks">
  <Row code="/^[A-Za-z][A-Za-z0-9_]*$/">validate an ASCII identifier in single-line input</Row>
  <Row code="/^\d+$/">validate one or more ASCII digits in single-line input</Row>
  <Row code="/^\p{Decimal_Number}+$/u">validate Unicode decimal digits in single-line input</Row>
  <Row code="/^\s*$/u">test whether the input is empty or entirely whitespace</Row>
  <Row code="/\r\n|[\n\r\u2028\u2029]/">match one ECMAScript line-break sequence</Row>
  <Row code="/[ \t]+$/gm">find trailing spaces or tabs on each line</Row>
  <Row code="/\bTODO\b/g">find `TODO` as a complete ASCII word</Row>
</Sheet>
