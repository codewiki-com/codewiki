<Sheet title="Declarations and inference">
  <Row code='const port: number = 8080'>give a binding an explicit value type</Row>
  <Row code='let result: string | undefined'>declare a value that may be undefined</Row>
  <Row code='const tags = ["ts", "types"]'>infer a mutable string array</Row>
  <Row code='const role = "admin" as const'>retain the exact string literal type</Row>
  <Row code='const pair = ["id", 42] as const'>infer a readonly literal tuple</Row>
  <Row code='const config = { mode: "prod" } satisfies AppConfig'>check the shape without replacing the inferred type</Row>
</Sheet>

<Sheet title="Objects and collections">
  <Row code='type User = { readonly id: string; name?: string }'>describe readonly and optional object properties</Row>
  <Row code='interface User { id: string; name?: string }'>declare an object contract that can be extended</Row>
  <Row code='type Pair = [label: string, value: number]'>label tuple positions for readable tooling</Row>
  <Row code='type Names = readonly string[]'>reject writes through an array type</Row>
  <Row code='type Dictionary = { [key: string]: number }'>describe unknown keys with one value type</Row>
  <Row code='type Entity = Identified & Timestamped'>combine two structural types</Row>
  <Row code='declare const UserIdBrand: unique symbol; type UserId = string & { readonly [UserIdBrand]: true }'>create a nominal string type with a unique brand</Row>
</Sheet>

<Sheet title="Function signatures">
  <Row code='function parse(text: string): number { return Number(text); }'>annotate parameter and return types</Row>
  <Row code='const handler: (event: Event) => void = (event) => event.preventDefault()'>contextually type a callback parameter</Row>
  <Row code='type Formatter = (value: number, unit?: string) => string'>mark one callback parameter optional</Row>
  <Row code='type Sum = (...values: number[]) => number'>type a rest parameter</Row>
  <Row code='interface Parse { (value: string): number; (value: number): string }'>describe overloaded call signatures</Row>
  <Row code='function handle(this: HTMLElement, event: Event): void {}'>declare the required call receiver</Row>
</Sheet>

<Sheet title="Unions and narrowing">
  <Row code='type Status = "idle" | "loading" | "done"'>restrict a value to listed literals</Row>
  <Row code='type Result = { ok: true; value: string } | { ok: false; error: Error }'>build a discriminated result union</Row>
  <Row code='typeof value === "string"'>narrow a primitive at runtime</Row>
  <Row code='"role" in account'>narrow by property presence</Row>
  <Row code='value instanceof Date'>narrow by a runtime constructor</Row>
  <Row code='const isString = (value: unknown): value is string => typeof value === "string"'>define a reusable type predicate</Row>
  <Row code='function assertString(value: unknown): asserts value is string { if (typeof value !== "string") throw new TypeError("Expected string"); }'>narrow after a validating assertion returns</Row>
</Sheet>

<Sheet title="Generic constraints">
  <Row code='function identity<T>(value: T): T { return value; }'>preserve the caller's input type</Row>
  <Row code='function first<T>(items: readonly T[]): T | undefined { return items[0]; }'>reflect that a collection may be empty</Row>
  <Row code='function get<T, K extends keyof T>(value: T, key: K): T[K] { return value[key]; }'>accept only keys present on the object</Row>
  <Row code='interface Box<T> { value: T }'>parameterize a reusable object contract</Row>
  <Row code='type ApiResult<T, E = Error> = { data: T } | { error: E }'>give a type parameter a default</Row>
  <Row code='class Store<T extends object> { constructor(readonly value: T) {} }'>constrain a class parameter to object types</Row>
</Sheet>

<Sheet title="Type queries">
  <Row code='type Config = typeof config'>derive a type from a value declaration</Row>
  <Row code='type ConfigKey = keyof Config'>collect an object's property keys as a union</Row>
  <Row code='type Mode = Config["mode"]'>look up one property type</Row>
  <Row code='type Item = (typeof items)[number]'>extract an array's element type</Row>
  <Row code='type ValueOf<T> = T[keyof T]'>collect an object's property value types</Row>
  <Row code='type Created = ReturnType<typeof createUser>'>extract a function's return type</Row>
  <Row code='type CreateArgs = Parameters<typeof createUser>'>extract parameters as a tuple</Row>
</Sheet>

<Sheet title="Mapped transformations">
  <Row code='type Optional<T> = { [K in keyof T]?: T[K] }'>make every property optional</Row>
  <Row code='type Mutable<T> = { -readonly [K in keyof T]: T[K] }'>remove readonly property modifiers</Row>
  <Row code='type Concrete<T> = { [K in keyof T]-?: T[K] }'>remove optional property modifiers</Row>
  <Row code='type Flags<T> = { [K in keyof T]: boolean }'>map every property to one value type</Row>
  <Row code='type Getters<T> = { [K in keyof T as `get${Capitalize<string & K>}`]: () => T[K] }'>remap property names into getter names</Row>
  <Row code='type StringsOnly<T> = { [K in keyof T as T[K] extends string ? K : never]: T[K] }'>keep only string-valued properties</Row>
</Sheet>

<Sheet title="Conditional types and infer">
  <Row code='type MessageOf<T> = T extends { message: unknown } ? T["message"] : never'>select a type when a property exists</Row>
  <Row code='type Element<T> = T extends readonly (infer U)[] ? U : never'>infer a readonly array's element type</Row>
  <Row code='type Head<T> = T extends readonly [infer H, ...unknown[]] ? H : never'>infer the first element of a tuple</Row>
  <Row code='type Unwrap<T> = T extends PromiseLike<infer U> ? Unwrap<U> : T'>recursively unwrap promise-like values</Row>
  <Row code='type Distributed<T> = T extends unknown ? T[] : never'>distribute over each union member</Row>
  <Row code='type NonDistributed<T> = [T] extends [unknown] ? T[] : never'>wrap the check to prevent distribution</Row>
</Sheet>

<Sheet title="Literal composition">
  <Row code='type Direction = "up" | "down"'>compose a finite string-literal union</Row>
  <Row code='type Route = `/users/${string}`'>require a fixed route prefix</Row>
  <Row code='type CssSize = `${number}px`'>require a numeric CSS pixel value</Row>
  <Row code='type EventName<K extends string> = `on${Capitalize<K>}`'>derive a capitalized event name</Row>
  <Row code='type EnvKey<K extends string> = `APP_${Uppercase<K>}`'>derive an uppercase environment key</Row>
  <Row code='type Position = `${"top" | "bottom"}-${"left" | "right"}`'>generate every union combination</Row>
</Sheet>

<Sheet title="Utility types">
  <Row code='Partial<User>'>make every property optional</Row>
  <Row code='Required<User>'>make every property required</Row>
  <Row code='Readonly<User>'>make every property readonly</Row>
  <Row code='Pick<User, "id" | "name"'>keep selected object properties</Row>

  <Row code='Record<Status, Handler>'>map every key to one value type</Row>

</Sheet>

<Sheet title="Modules and declarations">
  <Row code='import { value } from "./mod.js"'>import a runtime binding</Row>
  <Row code='import type { User } from "./user.js"'>import a type without a runtime dependency</Row>
  <Row code='export { value } from "./mod.js"'>re-export a runtime binding</Row>
  <Row code='export type { User } from "./user.js"'>re-export a type without runtime code</Row>
  <Row code='import data from "./data.json" with { type: "json" }'>use the current import-attribute syntax</Row>

  <Row code='export {}; declare global { interface Window { appVersion: string } }'>augment the global scope from a module</Row>
</Sheet>
