<Sheet title="Read repository state">
  <Row code="git status --short --branch">show the branch, upstream divergence, and compact working-tree and index status</Row>
  <Row code="git diff -- path/to/file">compare unstaged content with the index for one path</Row>
  <Row code="git diff --cached -- path/to/file">compare staged content with `HEAD` for one path</Row>
  <Row code="git diff HEAD -- path/to/file">compare working-tree content with `HEAD`, including staged and unstaged changes</Row>
  <Row code="git log --oneline --decorate --graph --all">draw the commit graph reachable from all local references</Row>
  <Row code="git show --stat --oneline COMMIT">inspect one commit's header and file-level change summary</Row>
</Sheet>

<Sheet title="Stage and commit">
  <Row code="git add -- path/to/file">copy the current path content into the index</Row>
  <Row code="git add -p -- path/to/file">choose patch hunks interactively for the index</Row>
  <Row code="git restore --staged -- path/to/file">reset the index path from `HEAD` while keeping the working-tree edit</Row>
  <Row code="git diff --cached --check">report whitespace errors in the change prepared for commit</Row>
  <Row code="git commit -m 'Explain the change'">commit the staged snapshot with a non-empty message</Row>
  <Row code="git commit --amend --no-edit">replace the latest commit with the current index while retaining its message</Row>
</Sheet>

<Sheet title="Branches and tags">
  <Row code="git switch -c feature/name main">create `feature/name` from `main` and switch to it</Row>
  <Row code="git switch main">switch the working tree and index to the local `main` branch</Row>
  <Row code="git branch --show-current">print the current branch name, or nothing in detached `HEAD` state</Row>
  <Row code="git branch -m feature/new-name">rename the current branch without changing its commits</Row>
  <Row code="git branch -d feature/name">delete a branch only when Git considers it fully merged</Row>
  <Row code="git tag -a v1.2.3 -m 'Release v1.2.3'">create an annotated tag at `HEAD`</Row>
  <Row code="git show --no-patch v1.2.3">inspect a tag and its target without printing the patch</Row>
</Sheet>

<Sheet title="Compare revisions">
  <Row code="git rev-parse --verify main^{commit}">resolve `main` and require it to identify a commit</Row>
  <Row code="git merge-base main feature/name">print a best common ancestor of two commit tips</Row>
  <Row code="git log main..feature/name --oneline">list commits reachable from the feature but not from `main`</Row>
  <Row code="git log --left-right --oneline main...feature/name">show commits unique to either side and mark which side contains each</Row>
  <Row code="git diff main...feature/name">compare the feature tip with its merge base against `main`</Row>
  <Row code="git diff COMMIT^ COMMIT --">show a non-root commit's change from its first parent</Row>
</Sheet>

<Sheet title="Fetch and inspect remotes">
  <Row code="git remote -v">show each remote's configured fetch and push URLs</Row>
  <Row code="git config --get-all remote.origin.fetch">inspect the refspecs used when fetching from `origin`</Row>
  <Row code="git fetch --prune origin">download from `origin` and remove stale remote-tracking references</Row>
  <Row code="git branch --remotes --verbose">list local remote-tracking references and their tip subjects</Row>
  <Row code="git log HEAD..origin/main --oneline">list fetched commits on `origin/main` that `HEAD` lacks</Row>
  <Row code="git diff --stat HEAD...origin/main">summarize the remote side's changes since the merge base</Row>
</Sheet>

<Sheet title="Integrate and publish">
  <Row code="git merge --ff-only origin/main">advance the current branch only when no merge commit is required</Row>
  <Row code="git merge --no-ff feature/name">merge the feature and create a merge commit even when fast-forwarding is possible</Row>
  <Row code="git pull --ff-only origin main">fetch remote `main` and accept only a fast-forward update</Row>
  <Row code="git push --dry-run origin HEAD">preview the remote reference update without changing it</Row>
  <Row code="git push -u origin feature/name">publish the local branch and record its upstream</Row>
  <Row code="git push --force-with-lease origin feature/name">force a rewritten branch only if the lease expectation still matches</Row>
  <Row code="git push --atomic origin main v1.2.3">require the branch and tag updates to succeed or fail together</Row>
</Sheet>

<Sheet title="Resolve merge conflicts">
  <Row code="git ls-files --unmerged">show the base, stage-2 side, and stage-3 side recorded for each conflict</Row>
  <Row code="git show :2:path/to/file">read the current operation's stage-2 version</Row>
  <Row code="git show :3:path/to/file">read the current operation's stage-3 version</Row>
  <Row code="git add -u -- path/to/file">replace a tracked path's unmerged index entries with its resolved state</Row>
  <Row code="git diff --check">find leftover conflict markers and whitespace errors before continuing</Row>
  <Row code="git merge --continue">finish the merge after resolving and staging every conflict</Row>
  <Row code="git merge --abort">restore the pre-merge state when the merge must be abandoned</Row>
</Sheet>

<Sheet title="Rebase and cherry-pick">
  <Row code="git rebase main">replay the current branch's unique commits onto `main`</Row>
  <Row code="git rebase -i HEAD~5">reorder, combine, edit, or drop the latest five commits</Row>
  <Row code="git rebase --onto main OLD_BASE feature/name">replay `OLD_BASE..feature/name` onto `main` and check out the rewritten feature</Row>
  <Row code="git rebase --continue">resume a stopped rebase after staging the resolution</Row>
  <Row code="git rebase --abort">return the branch and working tree to the pre-rebase state</Row>
  <Row code="git cherry-pick -x COMMIT">apply one commit's change and record its source identifier</Row>
  <Row code="git cherry-pick --abort">restore the state from before the current cherry-pick sequence</Row>
</Sheet>

<Sheet title="Undo by storage layer">
  <Row code="git restore -- path/to/file">discard unstaged tracked edits by copying the index version to the working tree</Row>
  <Row code="git restore --source=COMMIT -- path/to/file">replace a working-tree path from a commit without changing the index</Row>
  <Row code="git restore --source=HEAD --staged --worktree -- path/to/file">replace both index and working-tree versions of one path from `HEAD`</Row>
  <Row code="git reset --soft HEAD^">move `HEAD` to its first parent without changing the index or working tree</Row>
  <Row code="git reset --mixed HEAD^">move `HEAD`, reset the index to its parent, and leave the working tree unchanged</Row>
  <Row code="git reset --hard HEAD^">move `HEAD` and reset tracked index and working-tree content to its parent</Row>
  <Row code="git revert COMMIT">record the inverse of an ordinary commit; reverting a merge also requires `-m`</Row>
</Sheet>

<Sheet title="Stash and worktrees">
  <Row code="git stash push -u -m 'Context for later'">stash tracked and untracked changes with a searchable message</Row>
  <Row code="git stash list">list stash entries without applying them</Row>
  <Row code="git stash show --patch stash@{0}">inspect the complete patch stored in one stash entry</Row>
  <Row code="git stash apply stash@{0}">apply a stash while retaining the stash entry</Row>
  <Row code="git stash branch rescue/stashed stash@{0}">create a branch at the stash base, apply it, and drop it only on success</Row>
  <Row code="git worktree add -b hotfix/name ../repo-hotfix main">create a new branch and check it out in another working tree</Row>
  <Row code="git worktree list --porcelain">list worktrees in a stable, script-friendly format</Row>
</Sheet>

<Sheet title="Find regressions and recover">
  <Row code="git bisect start BAD GOOD">start a binary search between a known bad and known good commit</Row>
  <Row code="git bisect run ./test-regression.sh">classify revisions by exit status: 0 good, 125 skip, and other 1–127 values bad</Row>
  <Row code="git bisect reset">end bisection and return to the branch or commit checked out before it</Row>
  <Row code="git reflog --all --date=iso">inspect dated local movements of `HEAD` and other references</Row>
  <Row code="git show --stat --oneline HEAD@{1}">inspect a candidate reflog position before recovering it</Row>
  <Row code="git branch rescue/lost HEAD@{1}">make the verified reflog position reachable from a new branch</Row>
  <Row code="git fsck --unreachable">report currently unreachable objects that may contain lost commits</Row>
</Sheet>

<Sheet title="Inspect objects and references">
  <Row code="git cat-file -t OBJECT">print an object's type</Row>
  <Row code="git cat-file -p OBJECT">pretty-print an object's content according to its type</Row>
  <Row code="git ls-tree -r --name-only COMMIT">list every path in a commit snapshot recursively</Row>
  <Row code="git show COMMIT:path/to/file">read one file directly from a committed snapshot</Row>
  <Row code="git ls-files --stage">show each index entry's mode, object identifier, stage, and path</Row>
  <Row code="git for-each-ref --format='%(refname) %(objectname)' refs/heads/">list local branch references with their full target identifiers</Row>
  <Row code="git update-ref refs/heads/main NEW OLD">update `main` atomically only if it still points to `OLD`</Row>
</Sheet>
