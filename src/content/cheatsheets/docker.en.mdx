<Sheet title="Environment and context">
  <Row code="docker version">show client and connected engine versions</Row>
  <Row code="docker compose version">show the installed Compose plugin version</Row>
  <Row code="docker info">inspect engine, storage, network, and runtime configuration</Row>
  <Row code="docker context show">print the context used by default</Row>
  <Row code="docker context ls">list contexts and their daemon endpoints</Row>
  <Row code="docker context use production">make production the default context for later commands</Row>
</Sheet>

<Sheet title="Container lifecycle">
  <Row code="docker run --name catalog-api -d -p 127.0.0.1:3000:3000 catalog:1.4.0">create a detached container with a host-local port</Row>
  <Row code="docker ps --filter name=catalog-api">list running containers whose name matches catalog-api</Row>
  <Row code="docker ps -a --filter status=exited">list stopped containers with an exited state</Row>
  <Row code="docker stop --time 30 catalog-api">send the stop signal and allow thirty seconds before force</Row>
  <Row code="docker start catalog-api">restart the existing stopped container without recreating it</Row>
  <Row code="docker wait catalog-api">wait for the container to stop and print its exit code</Row>
  <Row code="docker rm catalog-api">remove the stopped container and its writable layer</Row>
</Sheet>

<Sheet title="Inspect and debug">

  <Row code="docker inspect --format '{{json .State.Health}}' catalog-api">print only the container's health state as JSON</Row>
  <Row code="docker top catalog-api">show host-visible processes running in the container</Row>
  <Row code="docker stats --no-stream catalog-api">take one CPU, memory, network, and I/O usage snapshot</Row>

  <Row code="docker cp catalog-api:/app/report.json ./report.json">copy one file from the container to the current directory</Row>

<Sheet title="Runtime isolation">
  <Row code="docker run --rm --user 10001:10001 catalog:1.4.0">override the image user with an explicit non-root UID and GID</Row>
  <Row code="docker run --rm --read-only --tmpfs /tmp catalog:1.4.0">make the root filesystem read-only with writable memory-backed temporary space</Row>
  <Row code="docker run --rm --cap-drop=ALL --cap-add=NET_BIND_SERVICE catalog:1.4.0">drop Linux capabilities, then restore only low-port binding</Row>
  <Row code="docker run --rm --security-opt no-new-privileges=true catalog:1.4.0">prevent the process from gaining privileges through exec</Row>
  <Row code="docker run --rm --memory 512m --cpus 1.0 catalog:1.4.0">limit container memory and CPU allocation</Row>
  <Row code="docker run --rm --pids-limit 200 catalog:1.4.0">cap the number of processes the container may create</Row>

<Sheet title="Compose service model">
  <Row code="build: { context: . }">build this service from the current Compose project context</Row>
  <Row code="ports: ['127.0.0.1:3000:3000']">publish port 3000 only on the host loopback address</Row>
  <Row code="depends_on: { db: { condition: service_healthy } }">wait for the db health check before creating this service</Row>
  <Row code="volumes: ['catalog-data:/var/lib/postgresql/data']">mount a named volume at the database data directory</Row>
  <Row code="read_only: true">make the service container's root filesystem read-only</Row>
  <Row code="tmpfs: [/tmp]">give the service writable temporary memory storage</Row>
  <Row code="init: true">run a small init process to forward signals and reap children</Row>
</Sheet>
