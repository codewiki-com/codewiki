<Sheet title="Bindings and branches">
  <Row code="name = value">bind or rebind a name without copying the object</Row>
  <Row code="first, *middle, last = values">unpack the middle items into a new list</Row>
  <Row code="result = value if condition else fallback">evaluate and return only the selected branch</Row>
  <Row code="if (match := pattern.search(text)):">evaluate once, then retain the match object</Row>
  <Row code="match command:">evaluate one subject against cases from top to bottom</Row>
  <Row code="case {'action': action, **rest}:">capture one mapping value and the unmatched keys</Row>
</Sheet>

<Sheet title="Collections">
  <Row code="items[start:stop:step]">select a sequence slice without changing the source</Row>
  <Row code="[project(item) for item in items if keep(item)]">filter and transform into a new list</Row>
  <Row code="{key(item): value(item) for item in items}">build a dictionary from generated key-value pairs</Row>
  <Row code="unique = set(items)">deduplicate hashable values without preserving input order</Row>
  <Row code="mapping.get(key, fallback)">return the fallback only when the key is absent</Row>

</Sheet>
